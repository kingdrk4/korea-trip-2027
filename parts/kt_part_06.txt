const DEFAULT_FX=0.00294483;
function getFxRate(){
 const saved=parseFloat(localStorage.getItem('krwMyrRate'));
 return Number.isFinite(saved)&&saved>0?saved:DEFAULT_FX;
}
function saveFxRate(){
 const v=parseFloat(document.getElementById('fxRateInput').value);
 if(Number.isFinite(v)&&v>0){
   localStorage.setItem('krwMyrRate',String(v));
   convertKRW(); convertMYR();
   alert('换汇率已保存并同步');
 }else{
   alert('请输入有效汇率');
 }
}
function convertKRW(){
 const n=parseFloat((document.getElementById('krwInput').value||'').replace(/,/g,''));
 const out=document.getElementById('krwResult');
 if(!Number.isFinite(n)){out.textContent='RM —';return;}
 out.textContent='RM '+(n*getFxRate()).toLocaleString(undefined,{maximumFractionDigits:2});
}
function convertMYR(){
 const n=parseFloat((document.getElementById('myrInput').value||'').replace(/,/g,''));
 const out=document.getElementById('myrResult');
 if(!Number.isFinite(n)){out.textContent='₩ —';return;}
 out.textContent='₩ '+Math.round(n/getFxRate()).toLocaleString();
}
