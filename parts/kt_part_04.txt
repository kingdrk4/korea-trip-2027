async function cloudPull(first=false){
 if(!ROOM_CODE){lockScreen();return}
 try{
  const r=await fetch(`${CLOUD_URL}/rest/v1/rpc/get_trip_state`,{method:'POST',headers:apiHeaders(),body:JSON.stringify({p_code:ROOM_CODE})});
  if(r.status===401||r.status===403){nativeRemoveItem.call(localStorage,ROOM_KEY_STORAGE);ROOM_CODE='';cloudReady=false;lockScreen();return}
  if(!r.ok)throw new Error('pull '+r.status);
  const remote=(await r.json())||{}; const local=localTripState();
  if(Object.keys(remote).length===0&&Object.keys(local).length){cloudReady=true;await cloudWrite();return}
  if(stable(remote)!==stable(local)){
   cloudApplying=true;
   Object.keys(local).forEach(k=>nativeRemoveItem.call(localStorage,k));
   Object.entries(remote).forEach(([k,v])=>nativeSetItem.call(localStorage,k,String(v)));
   cloudApplying=false;
   setSyncBadge('💗 收到另一台设备更新','ok');
   setTimeout(()=>location.reload(),250); return;
  }
  cloudReady=true;setSyncBadge('🔒 私人双人同步已连接','ok');
 }catch(e){cloudReady=false;setSyncBadge('☁️ 离线模式','bad')}
}

}
